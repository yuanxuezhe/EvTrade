## 功能概述

股票信息同步任务的生命周期管理（stock-info-crawler）：

- **启动**（`POST /api/sync/stocks`，admin only，202 Accepted）：后端自动获取全市场代码（sina 全量 + positions 持仓合并）→ `sync_manager.start(all_codes)` 创建后台 asyncio.Task
- **停止**（`DELETE /api/sync/stocks`，admin only）：set stop_event，任务在当前一只处理完后优雅退出
- **状态**（`GET /api/sync/stocks/status`，admin only）：返回 SyncTask.to_dict()（job_id/state/counters/进度/ETA/错误）
- **互斥**：任何时刻最多 1 个同步任务（asyncio.Lock + is_running 检查）；重复启动返 409
- **WS 推送**：每处理完一只，runner 经 progress_callback → manager 推 `sync_update` 通道 `type='stock_sync_progress'`；不推单只 stock_synced（前端 IndexedDB 负责缓存）
- **防反爬**：每只间隔 0.5s，全市场 ~5500 只约需 46 分钟；单只失败跳过计入 failed

### SyncTask 状态模型

```python
@dataclass
class SyncTask:
    job_id: str = uuid4()          # UUID 字符串
    state: str = "idle"            # idle / running / done / stopped / failed
    counters: Dict = {"total":0, "processed":0, "inserted":0, "updated":0, "skipped":0, "failed":0}
    current_code / current_name: Optional[str]
    started_at: Optional[float]    # time.time()
    elapsed_s: float = 0.0; eta_s: float = 0.0
    error_message: str = ""
    _task: Optional[asyncio.Task]           # 内部句柄, 不序列化
    _stop_event: Optional[asyncio.Event]

    def is_running(self): return self.state == "running"
    def to_dict(self): return {job_id, state, 6 个计数器拍平, current_code, current_name,
                               elapsed_s(round 1), eta_s(round 1), error_message}
```

## 核心实现

### 1. 启动端点与代码合并（api/sync.py）

```python
@router.post("/stocks", status_code=202, dependencies=[Depends(require_admin)])
async def start_sync_stocks():
    all_codes = await _get_all_stock_codes()
    if not all_codes:
        raise HTTPException(500, detail="no stock codes to sync")
    try:
        task = await sync_manager.start(all_codes)
        return {"code": 0, "msg": "started", "job_id": task.job_id, "total": task.counters["total"]}
    except RuntimeError as e:
        raise HTTPException(409, detail=str(e))          # 已有任务在跑

async def _get_all_stock_codes() -> List[str]:
    try:
        position_codes = [r.stock_code for r in Positions.query_all()]   # 已持仓兜底
    except Exception:
        position_codes = []
    from server.crawler.sources.sina_list import fetch_all_a_codes       # 全市场 ~5529 只
    try:
        sina_codes = fetch_all_a_codes(use_cache=True)                   # 24h 缓存
    except RuntimeError as exc:
        raise HTTPException(500, detail=f"全市场代码拉取失败 (sina_list): {exc}")
    return sorted(set(sina_codes) | set(position_codes))                 # 合并去重排序
```

合并逻辑原因：持仓代码兜底（交易过的小盘股可能不在 sina 当前列表）；builtin 常用列表被 sina 完全覆盖已删除。

### 2. SyncManager（services/sync/manager.py）

```python
class SyncManager:
    def __init__(self):
        self.current_task: Optional[SyncTask] = None
        self._lock = asyncio.Lock()                      # 防并发 start

    async def start(self, all_codes: List[str]) -> SyncTask:
        async with self._lock:
            if self.current_task is not None and self.current_task.is_running():
                raise RuntimeError(f"已有同步任务在跑: job_id={...}, processed={...}")
            task = SyncTask()
            task.state = "running"; task.started_at = time.time()
            task.counters["total"] = len(all_codes)
            task._stop_event = asyncio.Event()
            task._task = asyncio.ensure_future(self._run(task, all_codes))
            self.current_task = task                     # 覆盖旧（已结束）任务
            return task

    async def stop(self) -> bool:
        if self.current_task is None or not self.current_task.is_running(): return False
        self.current_task._stop_event.set()              # 优雅停止信号
        return True

    def status(self) -> Optional[dict]:
        if self.current_task is None: return None
        if self.current_task.is_running() and self.current_task.started_at:
            self.current_task.elapsed_s = time.time() - self.current_task.started_at   # 实时刷新
        return self.current_task.to_dict()

sync_manager = SyncManager()                             # 模块级全局单例
```

### 3. 后台执行 _run（progress_callback 桥接）

```python
async def _run(self, task, all_codes):
    try:
        def progress_callback(msg: dict):                # runner 每只调一次
            task.state = msg.get("state", task.state)
            task.current_code = msg.get("current_code"); task.current_name = msg.get("current_name")
            task.elapsed_s = msg.get("elapsed_s", 0); task.eta_s = msg.get("eta_s", 0)
            for k in ("total", "processed", "inserted", "updated", "skipped", "failed"):
                if k in msg: task.counters[k] = msg[k]
            asyncio.ensure_future(self._broadcast_progress(msg, task))   # fire-and-forget 推 WS
        await crawler_runner.run(all_codes=all_codes, progress_callback=progress_callback,
                                 stop_event=task._stop_event)
    except Exception as e:
        task.state = "failed"; task.error_message = str(e)
        await self._broadcast_progress({"type": "stock_sync_progress", "state": "failed",
                                        "error_message": str(e)}, task)
```

### 5. runner 同步循环（crawler/runner.py）

```python
async def run(all_codes, progress_callback, stop_event,
              sleep_sec=0.5, timeout_sec=10.0):
    counters = {"total": len(all_codes), "processed": 0, "inserted": 0,
                "updated": 0, "skipped": 0, "failed": 0}
    started_at = time.time()
    for stock_code in all_codes:
        if stop_event.is_set():                          # 每只开始前检查 → 优雅停止
            counters["state"] = "stopped"
            progress_callback({**counters, "current_code": None, "elapsed_s": ..., "eta_s": 0})
            return counters
        try:
            data = await asyncio.to_thread(eastmoney.fetch_base_info, stock_code, timeout_sec)
        except Exception:
            data = None                                  # 单只失败不中断
        action = "failed"; stock_name = ""
        if data is not None:
            stock_name = data.get("stock_name", "")
            db = SessionLocal()
            try:
                action = stocks_repo.upsert(db, stock_code, data)   # inserted/updated/skipped
            except Exception:
                action = "failed"
            finally:
                db.close()
        counters["processed"] += 1
        counters[action if action in ("inserted","updated","skipped") else "failed"] += 1
        elapsed = time.time() - started_at
        eta = (elapsed / max(counters["processed"], 1)) * (counters["total"] - counters["processed"])
        progress_callback({**counters, "state": "running", "current_code": stock_code,
                           "current_name": stock_name,
                           "elapsed_s": round(elapsed, 1), "eta_s": round(eta, 1)})
        await asyncio.sleep(sleep_sec)                   # 防反爬
    counters["state"] = "done"
    progress_callback({**counters, "current_code": None, "current_name": None,
                       "elapsed_s": ..., "eta_s": 0})
    return counters
```

设计要点：eastmoney.fetch_base_info 是同步函数，`asyncio.to_thread` 转异步；每只独立 DB session（finally close）；ETA = 平均耗时 × 剩余数。

### 6. 停止与状态端点

```python
@router.delete("/stocks", dependencies=[Depends(require_admin)])
async def stop_sync_stocks():
    stopped = await sync_manager.stop()
    if not stopped: raise HTTPException(404, detail="no running sync task")
    return {"code": 0, "msg": "stop requested"}          # 仅请求停止, 实际退出在当前只完成后

@router.get("/stocks/status", dependencies=[Depends(require_admin)])
async def get_sync_status():
    status = sync_manager.status()
    if status is None: return {"code": 0, "msg": "idle", "task": None}
    return {"code": 0, "msg": "ok", "task": status}
```

## 依赖关系

- **上游**：前端管理页（启动/停止/进度条）；全部端点 require_admin
- **下游**：crawler/runner.py → eastmoney.fetch_base_info（逐只拉）+ repo/stocks.upsert（入库）；sina_list.fetch_all_a_codes（代码清单，24h 缓存）
- **WS**：ws_manager.broadcast('sync_update')（前端订阅该频道渲染进度）
- **数据表**：stocks（upsert 目标）、positions（代码合并来源）
- **进程约束**：SyncManager/current_task 是进程内单例状态，与 auth session cache 相同——**单进程部署前提**

## 修改指南

- **调同步速度**：改 runner.py 的 `DEFAULT_SLEEP_SEC`（如 0.2），或 start 端点透传 sleep_sec 参数；过快有封 IP 风险
- **新增数据源字段**：改 eastmoney.fetch_base_info 返回 dict → repo/stocks.upsert 的写入字段 → stocks 表加列；runner 无需动（只透传 dict）
- **并发拉取**：runner 是刻意单线程顺序；若改并发，须保住「stop_event 每只前检查」「单只失败不中断」「progress_callback 计数正确」三契约，建议用 asyncio.Semaphore 限并发 + 每只间隔改为信号量节奏
- **任务持久化**（重启后可查历史）：当前 current_task 内存态，重启即失；如需历史需加表记录 job_id/counter 终态
- **允许并行多任务**：需把 current_task 单例改为 dict[job_id → SyncTask] + 启动端点去掉 409；WS 载荷已带 job_id 可区分
- **错误流细化**：runner 的 action 只有 inserted/updated/skipped/failed 四态；要区分超时/反爬/解析失败，在 fetch 异常分支给 msg 附 `error_kind` 字段并在前端展示
- **启动前 RPC/时段检查**：股票信息同步不依赖柜台 RPC，无需 require_rpc_ok；如要限制只在盘后跑，在 start 端点加 TradingClock 判断
